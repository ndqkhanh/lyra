⚡ Quickstart

Clone, install, set your API key. Lyra works with any LLM provider.

# Clone the repo git clone https://github.com/ndqkhanh/lyra && cd lyra # Set your API key export ANTHROPIC_API_KEY="sk-ant-..." # Install & run pip install -e "." lyra # Launch the TUI
# Clone the repo git clone https://github.com/ndqkhanh/lyra && cd lyra # Set your DeepSeek API key export DEEPSEEK_API_KEY="sk-..." # Install & run with DeepSeek pip install -e "." lyra --model deepseek-v4-pro
# Clone the repo git clone https://github.com/ndqkhanh/lyra && cd lyra # Set your OpenAI API key export OPENAI_API_KEY="sk-..." # Install & run with GPT pip install -e "." lyra --model gpt-5.1
# Clone the repo git clone https://github.com/ndqkhanh/lyra && cd lyra # Ensure Ollama is running with a model ollama pull qwen3:14b # Install & run locally pip install -e "." lyra --provider ollama --model qwen3:14b
